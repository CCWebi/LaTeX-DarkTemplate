\titleformat{\section}[frame]
    {\newpage\filcenter\bfseries\scshape\Large\color{Variante4}}
    { \ \thesection \ \ }
    {1ex}
    {\filcenter\LARGE}
    []

\titleformat{\subsection}[display]
    {\filcenter\bfseries\large\color{Variante5}}
    {}
    {16pt}
    {\thesubsection. \Large}
    [{\titlerule[1pt]}]

\titleformat{\subsubsection}[block]
    {\bfseries\normalsize\color{Variante6}}
    {\thesubsubsection .}
    {6pt}
    {\large}
    []

\titlespacing*{\subsection}{0pt}{0pt}{2ex plus .5ex}
\titlespacing*{\subsubsection}{0pt}{6ex plus 1ex minus .5ex}{1.5ex plus .5ex}
